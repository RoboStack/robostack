cmake_minimum_required(VERSION 3.16)
project(validator_check CXX)
find_package(nlohmann_json_schema_validator REQUIRED)
add_executable(validator_check main.cpp)
if(TARGET nlohmann_json_schema_validator::validator)
  target_link_libraries(validator_check nlohmann_json_schema_validator::validator)
else()
  target_link_libraries(validator_check nlohmann_json_schema_validator)
endif()
target_compile_features(validator_check PRIVATE cxx_std_17)
